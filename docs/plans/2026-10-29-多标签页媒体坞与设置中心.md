# 多标签页、媒体坞与设置中心 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 readbook 实现全隔离多标签页（插件视图多实例+文本快照）、全局媒体坞底部栏、设置中心弹窗三项需求（设计定稿见 `docs/多标签页、媒体坞与设置中心方案.md`）。

**Architecture:** 插件从模块级单例重构为实例工厂（`createView()`，旧接口插件自动包装保持兼容）；每个 tab = 文本层快照 + 可选插件视图实例（pane 保活，`display:none` 下视频可后台出声）；`app.state` 语义不变、始终代表激活 tab，切换即快照/恢复。媒体控件以 `MediaSession` 注册到全局底部栏，为音频插件预留。行为设置迁入设置中心弹窗。

**Tech Stack:** Tauri 2 + Vite + 原生 TypeScript（前端零框架）；Rust 侧仅 serde 设置结构扩展。

**测试约定:** 本仓库无测试基础设施（全 DOM 耦合，历来以 `npm run tauri dev` 手工验证）。每任务以 `npm run build`（tsc + vite 构建）为自动化门禁，附精确手工验证脚本。不引入 vitest（遵循项目现状）。

**约定:** 工作目录 `readbook/`；每任务一次 commit，信息末尾带 `Co-Authored-By: Claude Code <noreply@anthropic.com>`；手工验证用 `npm run tauri dev`（上级目录有 丰子恺散文.zip 等素材）。

- [ ] **Step 1.7: 新建 src/settings.ts**

创建 `src/settings.ts`，完整内容：

```ts
/* 设置中心（工具菜单「设置…」/ Ctrl+,）：行为类设置集中管理，即改即存。
   设计见 docs/多标签页、媒体坞与设置中心方案.md §3。
   副作用：keepAwake→常亮聚合器；naturalSort/含子目录→侧栏刷新；页模式间隔→reader。 */
import * as app from "./app";
import * as reader from "./reader";
import { setKeepAwakeEnabled } from "./keepawake";

export function isSettingsOpen(): boolean {
  const m = document.getElementById("settings-modal");
  return !!m && !m.classList.contains("hidden");
}

export function openSettings(): void {
  syncFromStore();
  document.getElementById("settings-modal")?.classList.remove("hidden");
}

export function closeSettings(): void {
  document.getElementById("settings-modal")?.classList.add("hidden");
}

function syncFromStore(): void {
  const s = app.state.store.settings;
  const set = (id: string, v: string | number | boolean) => {
    const el = document.getElementById(id) as HTMLInputElement | null;
    if (!el) return;
    if (el.type === "checkbox") el.checked = !!v;
    else el.value = String(v);
  };
  set("set-open-in-new-tab", s.openInNewTab);
  set("set-tab-switch-continue", s.tabSwitchPlayback === "continue");
  set("set-tab-limit", s.tabLimit);
  set("set-auto-restore", s.autoRestore);
  set("set-folder-auto-open", s.folderAutoOpen);
  set("set-natural-sort", s.naturalSort);
  set("set-folder-recursive", s.folderListRecursive);
  set("set-keep-awake", s.keepAwake);
  set("set-page-interval", s.pageInterval);
}

function bindCheck(id: string, apply: (v: boolean) => void): void {
  document.getElementById(id)?.addEventListener("change", (e) => {
    apply((e.target as HTMLInputElement).checked);
    app.persist();
  });
}

export function initSettingsModal(): void {
  const s = () => app.state.store.settings;
  bindCheck("set-open-in-new-tab", (v) => (s().openInNewTab = v));
  bindCheck("set-tab-switch-continue", (v) => (s().tabSwitchPlayback = v ? "continue" : "pause"));
  document.getElementById("set-tab-limit")?.addEventListener("change", (e) => {
    const v = Math.max(1, Math.min(50, Math.round(Number((e.target as HTMLInputElement).value)) || 20));
    s().tabLimit = v;
    (e.target as HTMLInputElement).value = String(v);
    app.persist();
  });
  bindCheck("set-auto-restore", (v) => (s().autoRestore = v));
  bindCheck("set-folder-auto-open", (v) => (s().folderAutoOpen = v));
  bindCheck("set-natural-sort", (v) => {
    s().naturalSort = v;
    void import("./folder").then((f) => f.refreshPanel());
    void import("./smb").then((m) => m.refreshAll());
  });
  bindCheck("set-folder-recursive", (v) => {
    s().folderListRecursive = v;
    void import("./folder").then((f) => f.refreshPanel());
  });
  bindCheck("set-keep-awake", (v) => {
    s().keepAwake = v;
    setKeepAwakeEnabled(v);
  });
  document.getElementById("set-page-interval")?.addEventListener("change", (e) => {
    const v = Math.max(1, Math.min(3600, Math.round(Number((e.target as HTMLInputElement).value)) || 10));
    s().pageInterval = v;
    (e.target as HTMLInputElement).value = String(v);
    reader.setPageInterval(v);
    app.persist();
  });
  document.getElementById("settings-close")?.addEventListener("click", closeSettings);
  document.getElementById("settings-done")?.addEventListener("click", closeSettings);
  document.getElementById("settings-modal")?.addEventListener("mousedown", (e) => {
    if (e.target === e.currentTarget) closeSettings();
  });
}
```

```ts
/** 插件视图实例表：多标签页下每个 tab 可持有一个视图（Task 3 接管生命周期归属） */
export interface ViewEntry {
  id: string; // 实例 id（常亮来源 view:<id> / 媒体坞会话 id）
  plugin: ReadBookPlugin;
  inst: PluginViewInstance;
  pane: HTMLElement; // .plugin-view-pane，宿主于 #plugin-view-host
  status: PluginViewStatus;
  file: string; // 当前查看的文件名（zip 条目为条目名）

  ctxPath: string; // 当前文件路径（插件跨容器推进随之变化）
  scopeZip: string;
  scopeDir: string;
}

const views = new Map<string, ViewEntry>();
let activeViewId = "";

export function activeEntry(): ViewEntry | null {
  return views.get(activeViewId) ?? null;
}
export function activeViewId(): string {
  return activeViewId;
}
export function instanceOf(id: string): PluginViewInstance | null {
  return views.get(id)?.inst ?? null;
}
export function viewPath(id: string): string {
```

```ts
function inScope(e: ViewEntry, path: string): boolean {
  const t = splitTarget(path);
  if (t.entry) return t.zipPath === e.scopeZip;
  if (e.scopeZip) return false; // 打开的是 zip 条目：不允许借道读外部磁盘文件
  return parentOf(t.base) === e.scopeDir;
}

async function loadBytesIn(e: ViewEntry, path: string): Promise<Uint8Array> {
  if (!inScope(e, path)) throw new Error(`插件只能访问当前文件所在目录/压缩包内的文件: ${path}`);
  const { zipPath, entry } = splitTarget(path);
  const buf = entry
    ? await api.readZipBytes(zipPath, entry)
    : await api.readFileBytes(path);
  return new Uint8Array(buf);
}

async function listContainerIn(e: ViewEntry): Promise<string[]> {
  try {
    if (e.scopeZip) return api.listZip(e.scopeZip, app.state.store.settings.naturalSort);
    const entries: FileEntry[] = await api.listDir(e.scopeDir, app.state.store.settings.naturalSort);
    return entries.filter((x) => !x.isDir).map((x) => x.path);
  } catch {
    return [];
  }
}
```

```ts
function snapshotFromState(): TextSnapshot | null {
  const s = app.state;
  if (!s.filePath) return null;
  return {
    filePath: s.filePath,
    realPath: s.realPath,
    zipEntry: s.zipEntry,
    isZip: s.isZip,
    fileName: s.fileName,
    encoding: s.encoding,
    ratio: plugins.isActive() ? 0 : reader.getRatio(),
    readSeconds: s.readSeconds,
    modified: s.modified,
    isHardLines: s.isHardLines,

function restoreStateFields(snap: TextSnapshot): void {
  const s = app.state;
  s.filePath = snap.filePath;
  s.realPath = snap.realPath;
  s.zipEntry = snap.zipEntry;
  s.isZip = snap.isZip;
  s.fileName = snap.fileName;
  s.encoding = snap.encoding;
  s.readSeconds = snap.readSeconds;
  s.modified = snap.modified;
  s.isHardLines = snap.isHardLines;
  s.raw = snap.raw ?? "";

/** 当前激活 tab 失活前的落账：快照 + 进度/时长写回历史（与关闭文件同语义） */
function storeLiveInto(tab: Tab): void {
  const snap = snapshotFromState();
  tab.text = snap;
  if (!snap) return;
  tab.title = snap.fileName;
  if (!plugins.isActive()) {
    const h = app.state.store.history.find((x) => x.filePath === snap.filePath);
    if (h) h.position = snap.ratio;
    app.state.store.settings.lastFilePath = snap.filePath;
    app.state.store.settings.lastScroll = snap.ratio;
  }
}

/** 播放策略：切走后按设置暂停后台媒体（"continue" 时不打断） */
function applyPlaybackPolicy(): void {

  for (const t of tabs) {
    if (t.id === activeId || !t.viewId) continue;
    const inst = plugins.instanceOf(t.viewId);
    try {
      if (inst?.isPlaying?.()) inst.togglePlay?.();
    } catch {
      /* 忽略 */
    }
  }

/* ========== 切换 / 新建 / 关闭 ========== */

export function switchToTab(id: string): void {
  if (id === activeId) return;
  const cur = activeTab();
  if (cur) storeLiveInto(cur);
  const target = tabs.find((t) => t.id === id);
  if (!target) return;
  activeId = id;
  activate(target);
  applyPlaybackPolicy();
  renderTabBar();
  app.persist();
}

function activate(target: Tab): void {
  reader.stopScroll();
  if (target.lazy) {
    const { path, ratio } = target.lazy;
    target.lazy = null;
    void app.openPath(path, { ratio, quiet: true, tab: "replace" });
    return; // openPath 完成内容装载与 UI 更新
  }
  if (target.viewId) {
    if (target.text) restoreStateFields(target.text); // 视图之下的文本层同步恢复（关视图回退用）
    plugins.activateView(target.viewId);
    return;
  }
  plugins.hideActiveView();
  if (target.text) {
    const snap = target.text;
    restoreStateFields(snap);
    if (snap.raw === null) {
      // 无驻留修改：重读文件（编码沿用快照，不重猜）；短暂显示旧内容属可接受的加载间隙
      void app.openPath(snap.filePath, {
        ratio: snap.ratio,
        encoding: snap.encoding || undefined,
        quiet: true,
        tab: "replace",
      });
    } else {
      app.renderContent(snap.ratio);
      (document.getElementById("reader-content") as HTMLElement).focus();
      app.updateUI();
    }
    return;
  }
  reader.showWelcome();
  app.updateUI();
}

export function createTab(title: string): Tab | null {
  const limit = Math.max(1, app.state.store.settings.tabLimit || 20);
  if (tabs.length >= limit) {
    app.toast(`标签页已达上限（${limit}），请先关闭部分标签页`);
    return null;
  }
  const cur = activeTab();
  if (cur) storeLiveInto(cur);
  const t: Tab = { id: crypto.randomUUID(), title, text: null, viewId: "", lazy: null, error: false };
  tabs.push(t);
  activeId = t.id;
  reader.stopScroll();
  plugins.hideActiveView(); // 旧 tab 的视图面板收起保活（归它自己的 tab）
  renderTabBar();
  return t;
}

export function closeTab(id: string): void {
  const idx = tabs.findIndex((t) => t.id === id);
  if (idx === -1) return;
  const tab = tabs[idx];
  if (tab.viewId) plugins.closeView(tab.viewId); // rb-view-closed → unbindView（幂等）
  tabs.splice(idx, 1);
  if (activeId === id) {
    activeId = "";
    const next = tabs[Math.min(idx, tabs.length - 1)];
    if (next) {
      switchToTab(next.id);
      applyPlaybackPolicy();
      return;
    }
    app.resetToWelcome();
  }
  renderTabBar();
  app.persist();
}

export function closeActiveTab(): void {
  if (activeId) closeTab(activeId);
}

/** 当前全部 tab 的持久化描述（app.persist 落盘前调用） */
export function descriptors(): TabDescriptorLite[] {

export function restoreFrom(ds: TabDescriptorLite[], activeIdx: number): void {
  for (const d of ds) {
    if (!d.path) continue;

      id: crypto.randomUUID(),

      error: false,
    });
  }

  if (tabs[i]) switchToTab(tabs[i].id);
  else renderTabBar();
```

```ts
    const t = tabs.createTab(viewName(path));
    if (!t) return; // 超出 tabLimit
  } else {
    tabs.destroyActiveView(); // 同 tab 换文件：旧视图销毁（翻页/惰性恢复语义）

  reader.stopScroll();

  try {
    if (zm) {
      // 条目扩展名命中启用插件（如图片）→ 插件接管
      if (plugins.handlerForExt(plugins.extOf(zm[2]))) {
        // 快照激活 tab 文本位置（插件视图覆盖后 reader 比例不可读）
        tabs.snapshotActiveRatio();
        const vid = await plugins.openFile(path);
        if (vid) tabs.bindView(vid);
        else tabs.markActiveError();
        return;
      }
      await openZipEntry(zm[1], zm[2], opts);
      return;
    }
    if (plugins.handlerForExt(plugins.extOf(path))) {
      tabs.snapshotActiveRatio();
      const vid = await plugins.openFile(path);
      if (vid) tabs.bindView(vid);
      else tabs.markActiveError();
      return;
    }
    const fc = await api.openText(path, opts.encoding);
    loadFileContent(fc, { isZip: false, zipPath: "", zipEntry: "" }, opts);
  } catch (e) {
    tabs.markActiveError();
    if (opts.quiet) {
      toast(`未能恢复上次阅读：${String(e).slice(0, 60)}`);
    } else {
      showMessage("打开失败", String(e));
    }
  }
}
```
