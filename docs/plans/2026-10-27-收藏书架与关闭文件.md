# 收藏书架增强与关闭文件 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 文本文件可关闭（逐层：插件视图→文本）；收藏书架升级为多书架（新建/重命名/删除/切换记忆），条目支持标签（多选并集筛选、命中数降序）与文件名搜索。设计全文见 `docs/收藏书架多书架与关闭文件方案.md`。

**Architecture:** 数据层 `Bookshelf { id, name, items }[]` 替换旧扁平 `bookshelf` 数组，条目嵌套归属书架（无 shelfId 外键）；旧数据在 Rust `load_store` 与前端 `mergeDefaults` 两处迁移（收进 id=`shelf-default` 的「默认书架」）；标签 `tags: string[]` 挂在条目上，`#[serde(default)]`/`|| []` 兼容旧 JSON。UI 集中在书架面板：header 下加书架行（chips 切换 + 行内新建/重命名 + 下拉确认删除），其下加搜索框与标签筛选 chips；条目行内展开标签编辑器。关闭文件为 `app.closeFile()` 单函数 + 菜单项 + Ctrl+W。

**Tech Stack:** Tauri 2 / Rust（serde）/ vanilla TypeScript，无前端测试 runner。

**测试约定:** 与既有计划相同——前端无 runner，每任务以 `npx tsc --noEmit` + 浏览器演示模式（`npm run dev`，mock 后端为内存 store）手测代替；Rust 侧用 `#[cfg(test)]` 单测（Task 1，TDD）。提交信息用中文、风格随仓库，末尾加 `Co-Authored-By: Claude Code <noreply@anthropic.com>`。

```rust
/// 旧版扁平收藏列表迁入默认书架（幂等：新列表非空时丢弃旧列表）
fn migrate_legacy_bookshelf(data: &mut StoreData) {
    if data.bookshelves.is_empty() && !data.bookshelf.is_empty() {
        data.bookshelves.push(Bookshelf {
            id: "shelf-default".to_string(),
            name: "默认书架".to_string(),
            items: std::mem::take(&mut data.bookshelf),
        });
    } else if !data.bookshelf.is_empty() {
        data.bookshelf.clear();
    }
}

pub fn load_store(app: &AppHandle) -> StoreData {
    match store_path(app) {
        Some(p) if p.exists() => {
            let data = fs::read_to_string(&p).unwrap_or_default();
            let mut store: StoreData = serde_json::from_str(&data).unwrap_or_default();
            migrate_legacy_bookshelf(&mut store);
            store
        }
        _ => StoreData::default(),
    }
}
```

```ts
/* ========== 关闭文件（逐层：插件视图 → 文本） ========== */

export function closeFile(): void {
  if (plugins.isActive()) {
    plugins.deactivateActive();
    // 底下无文本（比如启动后直接打开的图片/视频）→ 欢迎页
    if (!state.filePath) reader.showWelcome();
    updateUI();
    return;
  }
  if (!state.filePath) return;
  reader.stopScroll();
  saveCurrentProgress(); // 最终位置与时长写入历史
  state.filePath = "";
  state.realPath = "";
  state.zipEntry = "";
  state.isZip = false;
  state.fileName = "";
  state.raw = "";
  state.encoding = "";
  state.modified = false;
  state.isHardLines = false;
  state.prevPositions = [];
  state.readSeconds = 0;
  reader.showWelcome();
  // 清掉自动恢复锚点，否则下次启动又把这本书“复活”
  state.store.settings.lastFilePath = "";
  state.store.settings.lastScroll = 0;
  persist();
  updateUI();
}
```

```ts
/* 书架数据迁移/清洗（兼容旧形状：扁平 bookshelf、条目缺 tags、缺 bookshelves） */
function migrateBookshelves(data: StoreData): Bookshelf[] {
  const legacy = ((data as StoreData & { bookshelf?: BookshelfItem[] }).bookshelf || []).map((it) => ({
    ...it,
    tags: it.tags || [],
  }));
  let shelves = (data.bookshelves || []).map((s) => ({
    id: s.id || "shelf-default",
    name: s.name || "默认书架",
    items: (s.items || []).map((it) => ({ ...it, tags: it.tags || [] })),
  }));
  if (shelves.length === 0 && legacy.length > 0) {
    shelves = [{ id: "shelf-default", name: "默认书架", items: legacy }];
  }
  if (shelves.length === 0) {
    shelves = [{ id: "shelf-default", name: "默认书架", items: [] }];
  }
  return shelves;
}
```

```ts
/** 当前书架（activeId 失效时回退第一个） */
export function activeShelf(): Bookshelf {
  const shelves = state.store.bookshelves;
  return shelves.find((s) => s.id === state.store.settings.bookshelfActiveId) || shelves[0];

export function addToBookshelf(name?: string): void {
  // 插件视图（如图片查看）中收藏当前查看的文件（跨容器推进后取当前路径）
  const path = plugins.isActive() ? plugins.activeFilePath() : state.filePath;
  if (!path) return;
  const shelf = activeShelf();
  if (shelf.items.some((b) => b.filePath === path)) {
    toast("已在当前书架中");
    return;
  }
  shelf.items.unshift({
    filePath: path,
    name: name || viewName(path).replace(/\.txt$/i, ""),
    added: new Date().toISOString(),
    tags: [],
  });
  persist();
  toast(`已加入「${shelf.name}」`);
  document.dispatchEvent(new CustomEvent("rb-store-changed"));
}

export function addShelf(name: string): boolean {
  const n = name.trim();
  if (!n) return false;
  const shelf: Bookshelf = { id: crypto.randomUUID(), name: n, items: [] };
  state.store.bookshelves.push(shelf);
  state.store.settings.bookshelfActiveId = shelf.id; // 新建后立即切过去
  persist();
  document.dispatchEvent(new CustomEvent("rb-store-changed"));
  return true;
}

export function renameShelf(id: string, name: string): boolean {
  const n = name.trim();
  const shelf = state.store.bookshelves.find((s) => s.id === id);
  if (!shelf || !n) return false;
  shelf.name = n;
  persist();
  document.dispatchEvent(new CustomEvent("rb-store-changed"));
  return true;
}

export function removeShelf(id: string): void {
  const shelves = state.store.bookshelves;
  if (shelves.length <= 1) return; // 至少保留一个书架
  const idx = shelves.findIndex((s) => s.id === id);
  if (idx < 0) return;
  shelves.splice(idx, 1);
  if (state.store.settings.bookshelfActiveId === id) {
    state.store.settings.bookshelfActiveId = shelves[Math.max(0, idx - 1)].id;
  }
  persist();
  document.dispatchEvent(new CustomEvent("rb-store-changed"));
}

export function setActiveShelf(id: string): void {
  if (!state.store.bookshelves.some((s) => s.id === id)) return;
  state.store.settings.bookshelfActiveId = id;
  persist(); // 启动恢复当前书架
  document.dispatchEvent(new CustomEvent("rb-store-changed"));
}

/** 删除书架条目（filePath 定位，过滤态下索引不稳定故不用索引） */
export function deleteBookshelfItem(shelfId: string, filePath: string): void {
  const shelf = state.store.bookshelves.find((s) => s.id === shelfId);
  if (!shelf) return;
  const i = shelf.items.findIndex((b) => b.filePath === filePath);
  if (i < 0) return;
  shelf.items.splice(i, 1);
  persist();
  document.dispatchEvent(new CustomEvent("rb-store-changed"));
}

/** 设置条目标签（清洗：trim、截断 20 字符、去空、去重） */
export function setBookshelfItemTags(shelfId: string, filePath: string, tags: string[]): void {
  const shelf = state.store.bookshelves.find((s) => s.id === shelfId);
  if (!shelf) return;
  const item = shelf.items.find((b) => b.filePath === filePath);
  if (!item) return;
  const cleaned: string[] = [];
  for (const t of tags) {
    const v = t.trim().slice(0, 20);
    if (v && !cleaned.includes(v)) cleaned.push(v);
  }
  item.tags = cleaned;
  persist();
  document.dispatchEvent(new CustomEvent("rb-store-changed"));
}

/** 全部书架的标签并集（编辑建议词表） */
export function allShelfTags(): string[] {
  const set = new Set<string>();
  for (const s of state.store.bookshelves) for (const it of s.items) for (const t of it.tags) set.add(t);
  return [...set];
}
```

```ts
/* 书架条目行：名称 + 标签 chips + 🏷 编辑 + ✕ 删除（不用通用 rowEl，需塞标签） */
function renderBookshelfItemRow(item: BookshelfItem): HTMLElement {
  const shelfId = app.activeShelf().id;
  const row = document.createElement("div");
  row.className = "list-item";
  const info = document.createElement("div");
  info.className = "list-item-info";
  const n = document.createElement("div");
  n.className = "item-name";
  n.textContent = item.name;
  const d = document.createElement("div");
  d.className = "item-desc";
  d.textContent = `加入于 ${fmtTime(item.added)}`;
  info.append(n, d);
  if (item.tags.length > 0) {
    const tagRow = document.createElement("div");
    tagRow.className = "bs-tags";
    item.tags.forEach((t) => {
      const c = document.createElement("span");
      c.className = "bs-tag";
      c.textContent = t;
      tagRow.appendChild(c);
    });
    info.appendChild(tagRow);
  }
  info.addEventListener("click", () => app.openPath(item.filePath));
  row.appendChild(info);
  const tagBtn = document.createElement("button");
  tagBtn.className = "item-tag-edit";
  tagBtn.textContent = "🏷";
  tagBtn.title = "编辑标签";
  tagBtn.addEventListener("click", (e) => {
    e.stopPropagation();
    tagEditPath = tagEditPath === item.filePath ? null : item.filePath;
    renderBookshelfList();
  });
  row.appendChild(tagBtn);
  const del = document.createElement("button");
  del.className = "item-delete";
  del.textContent = "✕";
  del.title = "删除";
  del.addEventListener("click", (e) => {
    e.stopPropagation();
    app.deleteBookshelfItem(shelfId, item.filePath);
  });
  row.appendChild(del);
  return row;
}

/* 行内标签编辑器：输入框（回车/逗号提交，可一次多个）+ 全局标签建议 chips */
function buildTagEditor(item: BookshelfItem): HTMLElement {
  const shelfId = app.activeShelf().id;
  const box = document.createElement("div");
  box.className = "bs-tag-editor";
  const input = document.createElement("input");
  input.type = "text";
  input.placeholder = "输入标签，回车添加（可逗号/空格分隔多个）";
  const close = document.createElement("button");
  close.className = "bs-tag-close";
  close.textContent = "✕";
  close.title = "收起";
  const chips = document.createElement("div");
  chips.className = "bs-tag-editor-chips";
  const setTags = (tags: string[]) => {
    app.setBookshelfItemTags(shelfId, item.filePath, tags);
    renderBookshelf(); // 刷新行内 chips、编辑器开关态与筛选行
  };
  const addFromInput = () => {
    const parts = input.value.split(/[,，、\s]+/).filter(Boolean);
    if (parts.length) setTags([...item.tags, ...parts]);
    else input.focus();
  };
  input.addEventListener("keydown", (e) => {
    e.stopPropagation(); // 防止 Esc 顺带关掉面板
    if (e.key === "Enter") {
      e.preventDefault();
      addFromInput();
    }
    if (e.key === "Escape") {
      tagEditPath = null;
      renderBookshelf();
    }
  });
  close.addEventListener("click", () => {
    tagEditPath = null;
    renderBookshelf();
  });
  app.allShelfTags().forEach((t) => {
    const on = item.tags.includes(t);
    const c = document.createElement("button");
    c.className = "bs-tag-suggest" + (on ? " on" : "");
    c.textContent = (on ? "✓ " : "＋ ") + t;
    c.addEventListener("click", () => {
      setTags(on ? item.tags.filter((x) => x !== t) : [...item.tags, t]);
    });
    chips.appendChild(c);
  });
  const inputRow = document.createElement("div");
  inputRow.className = "bs-tag-input-row";
  inputRow.append(input, close);
  box.append(inputRow, chips);
  return box;
}

function renderBookshelfList(): void {
  const list = $("bookshelf-list");
  list.innerHTML = "";
  const shelf = app.activeShelf();
  if (shelf.items.length === 0) {
    const empty = document.createElement("div");
    empty.style.cssText = "padding:16px;font-size:12px;color:#888";
    empty.textContent = `「${shelf.name}」空空如也。打开一本书后点「添加当前」收进来。`;
    list.appendChild(empty);
    return;
  }
  shelf.items.forEach((b) => {
    list.appendChild(renderBookshelfItemRow(b));
    if (tagEditPath === b.filePath) list.appendChild(buildTagEditor(b));
  });
}
```

```ts
/* 标签筛选行：当前书架出现过的标签（多选并集），「全部」一键清除 */
function renderTagFilter(): void {
  const box = $("bookshelf-tag-filter");
  box.innerHTML = "";
  const tags = [...new Set(app.activeShelf().items.flatMap((it) => it.tags))];
  if (tags.length === 0) {
    box.classList.add("hidden");
    return;
  }
  box.classList.remove("hidden");
  const all = document.createElement("button");
  all.className = "bs-tag-filter-chip" + (bsTagFilter.size === 0 ? " on" : "");
  all.textContent = "全部";
  all.addEventListener("click", () => {
    bsTagFilter = new Set();
    renderBookshelf();
  });
  box.appendChild(all);
  tags.forEach((t) => {
    const on = bsTagFilter.has(t);
    const c = document.createElement("button");
    c.className = "bs-tag-filter-chip" + (on ? " on" : "");
    c.textContent = t;
    c.title = on ? "取消筛选该标签" : "按该标签筛选";
    c.addEventListener("click", () => {
      if (bsTagFilter.has(t)) bsTagFilter.delete(t);
      else bsTagFilter.add(t);
      renderBookshelf();
    });
    box.appendChild(c);
  });
}

/* 过滤 + 排序：命中任一选中标签即显示（并集），按命中标签数降序（sort 稳定，同数保持原顺序） */
function visibleBookshelfItems(): { item: BookshelfItem; hits: number }[] {
  const shelf = app.activeShelf();
  const out: { item: BookshelfItem; hits: number }[] = [];
  for (const it of shelf.items) {
    const nameOk =
      !bsSearch || it.name.toLowerCase().includes(bsSearch) || it.filePath.toLowerCase().includes(bsSearch);
    if (!nameOk) continue;
    if (bsTagFilter.size === 0) {
      out.push({ item: it, hits: 0 });
      continue;
    }
    const hits = it.tags.filter((t) => bsTagFilter.has(t)).length;
    if (hits > 0) out.push({ item: it, hits });
  }
  if (bsTagFilter.size > 0) out.sort((a, b) => b.hits - a.hits);
  return out;
}

function renderBookshelfList(): void {
  const list = $("bookshelf-list");
  list.innerHTML = "";
  const shelf = app.activeShelf();
  if (shelf.items.length === 0) {
    const empty = document.createElement("div");
    empty.style.cssText = "padding:16px;font-size:12px;color:#888";
    empty.textContent = `「${shelf.name}」空空如也。打开一本书后点「添加当前」收进来。`;
    list.appendChild(empty);
    return;
  }
  const visible = visibleBookshelfItems();
  if (visible.length === 0) {
    const empty = document.createElement("div");
    empty.style.cssText = "padding:16px;font-size:12px;color:#888";
    empty.textContent = "没有匹配的收藏，试试调整搜索词或标签筛选。";
    list.appendChild(empty);
    return;
  }
  visible.forEach(({ item }) => {
    list.appendChild(renderBookshelfItemRow(item));
    if (tagEditPath === item.filePath) list.appendChild(buildTagEditor(item));
  });
}
```
