**Goal:** 把窗口内拖拽打开做扎实（多文件、内容把关、悬停提示），新增「系统打开请求」统一入口，并给标签页加右键菜单（复制文件路径 / 在文件夹栏中定位 / 关闭标签页）。

**Architecture:** 拖拽与系统打开请求共用 `app.openExternalPaths`（一批路径的打开规则只写一处）。系统打开请求在 Rust 侧新增 `sysopen.rs`：启动参数、单实例插件转发的第二实例参数、macOS「打开文件」事件汇入同一个排队状态，前端就绪前排队、就绪后直发事件；`lib.rs` 由 `.run(ctx)` 改为 `.build(ctx).run(回调)`（`Builder::run` 会丢弃所有运行事件）。标签右键菜单放在 `tabs.ts`，定位复用 `folder.revealFile` / `smb.revealRemote`，只补失败提示。

**Tech Stack:** 原生 TypeScript（无前端框架）、Rust + Tauri 2.12.5。前端无测试设施：每个行为先写浏览器验收脚本（py-playwright + `npm run dev` 的演示后端）并看着它失败，再实现；Rust 侧用 `cargo test` 严格 TDD。

**设计文档:** `GLOSSARY.md`（拖拽打开 / 系统打开请求 / 文件夹栏定位）；共识清单见 2026-11-25 的 grill 会话结论。

# 系统打开请求 Linux 端到端（真实二进制）：冷启动带文件参数、已运行时第二实例转发、重启先恢复再打开。
# 用法：e2e_linux.sh <二进制路径>
# 外层建隔离的临时数据目录并开私有 D-Bus 会话；内层在该总线的同一 Xvfb 显示下跑实例。
set -u
APP="$1"; W=/tmp/rb-e2e
if [ "${2:-}" = "--inner" ]; then
  export XDG_DATA_HOME="$W/data" XDG_CONFIG_HOME="$W/config" XDG_CACHE_HOME="$W/cache"
  export WEBKIT_DISABLE_DMABUF_RENDERER=1 WEBKIT_DISABLE_COMPOSITING_MODE=1
  export STORE="$W/data/com.zhuangbin.read4ever/read4ever.json"
  tabs() { python3 - "$STORE" <<'PY'
import json, sys
try:
    d = json.load(open(sys.argv[1]))
    print(" | ".join(t["path"].rsplit("/", 1)[-1] for t in d["settings"]["tabs"]))
except Exception:
    print("<无法读取>")
PY
  }
  wait_store() { for _ in $(seq 1 240); do grep -q "$1" "$STORE" 2>/dev/null && return 0; sleep 0.5; done; return 1; }

  xvfb-run -a -s "-screen 0 1280x800x24" bash -c '
    run_and_wait() { # $1 文件名；启动一个实例并等它的路径落到存档里，返回实例 pid（全局 PID）
      "$0" "$1/books/$2" >/dev/null 2>&1 &
      PID=$!
      wait_store "$2"
    }
    if run_and_wait "$1" 冷启动.txt; then echo "PASS  冷启动：启动后打开了参数里的文件  tabs=[$(tabs)]"; else echo "FAIL  冷启动未打开文件  tabs=[$(tabs)]"; fi
    "$0" "$1/books/转发.txt" --not-a-file-flag >/dev/null 2>&1; RC=$?
    if wait_store 转发.txt && [ "$RC" = 0 ]; then echo "PASS  已运行：第二实例把文件转交给第一实例  tabs=[$(tabs)]"; else echo "FAIL  转发失败 rc=$RC tabs=[$(tabs)]"; fi
    if grep -q "not-a-file-flag" "$STORE" 2>/dev/null; then echo "FAIL  非路径参数被当成文件"; else echo "PASS  非路径参数被忽略"; fi

    if run_and_wait "$1" 第三个.txt && [ "$(tabs)" = "冷启动.txt | 转发.txt | 第三个.txt" ]; then echo "PASS  重启：先恢复上次的标签页再打开新文件  tabs=[$(tabs)]"; else echo "FAIL  重启后的标签页不对  tabs=[$(tabs)]"; fi

printf '转发\n' > "$W/books/转发.txt"

- [x] **Step 3: 新增 `openExternalPaths`**（放在 `openPath` 之后、`openZipEntry` 之前）

```ts
/* ========== 外部批量打开（拖拽 / 系统打开请求共用） ========== */

/** 打开一批来自外部的路径。分流与手动打开一致（openInNewTab、同文件已开则跳转、标签页上限）。
 *  可打开的按给定顺序依次打开，最后打开的成为当前标签页；打不开的（文件夹、无扩展名、
 *  陌生扩展名）跳过并轻提示，不建标签页。多条目 zip 排在其余文件之后逐个弹选择。 */
export async function openExternalPaths(paths: string[]): Promise<void> {
  const list = [...new Set(paths.filter(Boolean))];
  if (!list.length) return;
  await tabs.settleActivate();
  const skipped: string[] = [];
  const zips: string[] = [];
  const plain: string[] = [];
  for (const p of list) {
    if (!plugins.canOpenFile(baseName(p))) skipped.push(p);
    else if (p.toLowerCase().endsWith(".zip")) zips.push(p);
    else plain.push(p);
  }
  const limit = Math.max(1, state.store.settings.tabLimit || 20);
  const rest = [...plain, ...zips];
  let i = 0;
  for (; i < rest.length; i++) {
    const p = rest[i];
    // 上限只拦「要新开标签页」的：同文件已开的照常跳转已有标签页
    if (tabs.count() >= limit && !tabs.tabForPath(p)) break;
    try {
      await openPath(p, { batch: true });
    } catch (e) {
      showMessage("打开失败", String(e)); // zip 目录读不出来等：与非 zip 路径的失败提示一致
    }
  }
  const notes: string[] = [];
  if (i < rest.length) notes.push(`标签页已达上限（${limit}），还有 ${rest.length - i} 个未打开`);
  if (skipped.length) {
    const names = skipped.slice(0, 3).map(baseName).join("、");
    notes.push(`已跳过 ${skipped.length} 个无法打开的项：${names}${skipped.length > 3 ? " 等" : ""}`);
  }
  if (notes.length) toast(notes.join("；"));
}
```

```ts
/** 远程文件在系统里可直接使用的地址（「复制文件路径」用）。规范路径 smb://<服务器id>/<共享>/<路径>
 *  里的 id 是应用内部编号：主机名取自服务器配置，Windows 给 UNC 路径，其余给 smb URL。
 *  UNC 带不了端口（自定义端口的服务器在 Windows 上只能按主机名给）；服务器配置已删除时原样返回 */
export function systemPath(path: string): string {
  const segs = path.split("::")[0].replace(/^smb:\/*/, "").split("/").filter(Boolean);
  const srv = segs.length >= 2 ? servers().find((s) => s.id === segs[0]) : undefined;
  if (!srv) return path;
  const rest = segs.slice(1); // 共享名及其下的路径
  if (/Windows/i.test(navigator.userAgent)) {
    return `\\\\${srv.host.replace(/:\d+$/, "")}\\${rest.join("\\")}`;
  }
  return `smb://${srv.host}/${rest.join("/")}`;
}
```
